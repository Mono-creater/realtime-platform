   690	// ---------- 故障分析逻辑（阈值单一来源：plc-packet.js THRESHOLDS） ----------
   691	function analyzeFault(data) {
   692	    const faults = [];
   693	    if (data.temperature > THRESHOLDS.tempHigh) {
   694	        faults.push({
   695	            type: '高温预警',
   696	            level: 'warning',
   697	            value: data.temperature,
   698	            detail: `均值温度 ${data.temperature}°C > ${THRESHOLDS.tempHigh}°C`
   699	        });
   700	    } else if (data.temperature < THRESHOLDS.tempLow) {
   701	        faults.push({
   702	            type: '低温预警',
   703	            level: 'warning',
   704	            value: data.temperature,
   705	            detail: `均值温度 ${data.temperature}°C < ${THRESHOLDS.tempLow}°C`
   706	        });
   707	    }
   708	    if (data.pressure > THRESHOLDS.pressHigh) {
   709	        faults.push({
   710	            type: '高压报警',
   711	            level: 'error',
   712	            value: data.pressure,
   713	            detail: `均值压力 ${data.pressure}kPa > ${THRESHOLDS.pressHigh}kPa`
   714	        });
   715	    } else if (data.pressure < THRESHOLDS.pressLow) {
   716	        faults.push({
   717	            type: '低压报警',
   718	            level: 'error',
   719	            value: data.pressure,
   720	            detail: `均值压力 ${data.pressure}kPa < ${THRESHOLDS.pressLow}kPa`
   721	        });
   722	    }
   723	    if (data.humidity > THRESHOLDS.humidHigh) {
   724	        faults.push({
   725	            type: '高湿预警',
   726	            level: 'warning',
   727	            value: data.humidity,
   728	            detail: `均值湿度 ${data.humidity}% > ${THRESHOLDS.humidHigh}%`
   729	        });
   730	    }
   731	    return faults;
   732	}

   753	const lastAlarmAt = {};

   755	async function handleAlarm(faults, avg) {
   756	    for (const fault of faults) {
   757	        const prev = lastAlarmAt[fault.type];
   758	        const value = Number(fault.value) || 0;
   759	        if (prev && Date.now() - prev.time < 10000
   760	            && Math.abs(prev.value - value) <= Math.max(5, Math.abs(prev.value) * 0.05)) {
   761	            continue; // 重复告警，跳过
   762	        }
   763	        lastAlarmAt[fault.type] = { time: Date.now(), value: value };
   764	        const warning = {
   765	            code: 'PLC_SENSOR',
   766	            content: fault.type,
   767	            time: new Date().toISOString(),
   768	            worker: 'PLC_AUTO',
   769	            // 环境类预警备注写明具体成因（含触发数据），其余类型保留调试 JSON
   770	            remark: ENV_WARNING_CAUSES[fault.type]
   771	                ? `${ENV_WARNING_CAUSES[fault.type]}；触发数据：${fault.detail}`
   772	                : JSON.stringify({
   773	                    avg: avg,
   774	                    faults: faults,
   775	                    timestamp: new Date().toISOString()
   776	                })
   777	        };
   778	        const created = await saveWarningToHistory(warning);
   779	        const memWarning = {
   780	            id: created.id,
   781	            code: warning.code,
   782	            content: warning.content,
   783	            time: warning.time,
   784	            worker: warning.worker,
   785	            remark: warning.remark
   786	        };
   787	        pushWarning(memWarning);
   788	    }
   789	
   790	    updateOverLimitStats();
   791	    broadcastFullUpdate();
   792	}

   794	// ---------- Modbus 请求互斥锁（FC03 读 / FC16 写串行化） ----------
   795	let modbusChain = Promise.resolve();
   796	function modbusEnqueue(fn) {
   797	    const p = modbusChain.then(fn, fn);
   798	    modbusChain = p.catch(() => {});
   799	    return p;

   802	// ---------- 控制引擎（双环 PID + Z-N 整定） ----------
   803	const controlEngine = createControlEngine();
   804	let latestSensor = null;        // 最近一次传感器均值（喂给控制引擎）
   805	let lastPacket = null;          // 最近一次原始报文（/api/plc/packet 调试用）
   806	let controlLoopInterval = null;
   807	let lastWrittenForce = null;    // 写回去重
   808	let lastWrittenHeight = null;
   809	let lastWriteAt = 0;            // 最近写回时间：输出不变时每 5s 保活重写（ZN 整定期间输出恒定，防止 PLC 侧写回看门狗/模拟器漂移）
   810	
   811	function startControlLoop() {
   812	    if (controlLoopInterval) return;
   813	    controlLoopInterval = setInterval(() => {
   814	        const mode = SIMULATION_MODE ? 'simulation' : 'plc';
   815	        controlEngine.tick(latestSensor, mode);
   816	
   817	        // 写回：自动调节或整定中的回路，输出变化时 FC16 写 30021/30022；
   818	        // 输出不变则每 5s 保活重写一次（整定阶跃期间输出恒定，防止 PLC 侧看门狗判超时）
   819	        if (!SIMULATION_MODE && plcClient) {
   820	            const st = controlEngine.getStatus();
   821	            const tuning = st.tune && st.tune.state === 'sampling';
   822	            const forceActive = st.loops.force.auto || (tuning && st.tune.loop === 'force');
   823	            const postureActive = st.loops.posture.auto || (tuning && st.tune.loop === 'posture');
   824	            const forceRaw = Math.round(Math.min(1000, Math.max(50, st.loops.force.output)) * 10); // 压力寄存器 ×10（1000kPa→10000，uint16 安全）
   825	            const heightRaw = Math.round(Math.min(2600, Math.max(1000, st.loops.posture.output)));
   826	            const anyActive = forceActive || postureActive;
   827	            const forceChanged = forceRaw !== lastWrittenForce;
   828	            const heightChanged = heightRaw !== lastWrittenHeight;
   829	            const keepAlive = anyActive && Date.now() - lastWriteAt > 5000;
   830	            if (anyActive && (forceChanged || heightChanged || keepAlive)) {
   831	                if (forceChanged) lastWrittenForce = forceRaw;
   832	                if (heightChanged) lastWrittenHeight = heightRaw;
   833	                lastWriteAt = Date.now();
   834	                modbusEnqueue(() => plcClient.writeRegisters(
   835	                    PLC_CONFIG.registers.writeForce, [forceRaw, heightRaw]))
   836	                    .catch((err) => console.error('⚠️ 写回 PLC 失败（不影响采集）:', err.message));
   837	            }
   838	        }
   839	    }, controlEngine.config.controlInterval);
   840	}

   860	async function readAndBufferSensorData() {
   861	    if (!plcClient) {
   862	        if (Date.now() < plcRetryAt) return; // 退避等待中，跳过本次重连
   863	        plcClient = await connectPLC();
   864	        if (!plcClient) {
   865	            plcReconnectDelay = Math.min(plcReconnectDelay * 2, PLC_RECONNECT_MAX_MS);
   866	            plcRetryAt = Date.now() + plcReconnectDelay;
   867	            console.warn(`⚠️ PLC 连接失败，${(plcReconnectDelay / 1000).toFixed(0)}s 后重试`);
   868	            return;
   869	        }
   870	    }
   871	
   872	    try {
   873	        // 主包：30001-30005（地址连续，一次批量读取）
   874	        const regs = await modbusEnqueue(() =>
   875	            plcClient.readHoldingRegisters(PLC_CONFIG.registers.temperature, 5)
   876	                .then((r) => r.data));
   877	        const packet = parseMainPacket(regs);
   878	        plcReconnectDelay = PLC_RECONNECT_INITIAL_MS; // 读取成功，重置退避
   879	
   880	        if (packet.extended) {
   881	            // ===== 新固件扩展协议：直接用 PLC 侧 10 条均值，不再二次平均 =====
   882	            const avg = {
   883	                temperature: Math.round(packet.temperature * 10) / 10,
   884	                pressure: Math.round(packet.pressure * 100) / 100,
   885	                humidity: Math.round(packet.humidity * 10) / 10
   886	            };
   887	            latestSensor = avg;
   888	
   889	            let sub = null;
   890	            let faults = [];
   891	            if (packet.subValid) {
   892	                const subRegs = await modbusEnqueue(() =>
   893	                    plcClient.readHoldingRegisters(PLC_CONFIG.registers.subBase, 5)
   894	                        .then((r) => r.data));
   895	                sub = parseSubPacket(subRegs);
   896	                faults = anomaliesFromBitmap(sub.bitmap || packet.anomalyBitmap, sub.values);
   897	            } else if (packet.anomalyBitmap) {
   898	                faults = anomaliesFromBitmap(packet.anomalyBitmap, {
   899	                    temperature: packet.temperature,
   900	                    pressure: packet.pressure,
   901	                    humidity: packet.humidity
   902	                });
   903	            }
   904	            lastPacket = { source: 'plc', main: packet, sub, receivedAt: Date.now() };
   905	            broadcastSensorData(avg, faults, {
   906	                source: 'plc', anomalies: faults, packet: lastPacket
   907	            });
   908	            if (faults.length > 0) {
   909	                await handleAlarm(faults, avg);
   910	                console.log(`🚨 [PLC] seq=${packet.seq} 检测到 ${faults.length} 个异常（子包序号=${sub ? sub.subSeq : '-'}）`);
   911	            }
   912	        } else {
   913	            // ===== 旧固件回退：平台侧 10 条缓冲均值 =====
   914	            const record = {
   915	                timestamp: Date.now(),
   916	                temperature: Math.round(packet.temperature * 10) / 10,
   917	                pressure: Math.round(packet.pressure * 100) / 100,
   918	                humidity: Math.round(packet.humidity * 10) / 10
   919	            };
   920	            sensorBuffer.push(record);
   921	            if (sensorBuffer.length >= PLC_CONFIG.windowSize) {
   922	                await processBuffer();
   923	            }
   924	        }
   925	    } catch (err) {
   926	        console.error('❌ 读取传感器数据失败:', err.message);
   927	        plcClient = null;
   928	        plcReconnectDelay = Math.min(plcReconnectDelay * 2, PLC_RECONNECT_MAX_MS);
   929	        plcRetryAt = Date.now() + plcReconnectDelay;
   930	    }
   931	}

   933	async function processBuffer() {
   934	    if (sensorBuffer.length === 0) return;
   935	    const sum = sensorBuffer.reduce((acc, cur) => {
   936	        acc.temp += cur.temperature;
   937	        acc.press += cur.pressure;
   938	        acc.humid += cur.humidity;
   939	        return acc;
   940	    }, { temp: 0, press: 0, humid: 0 });
   941	    const count = sensorBuffer.length;
   942	    const avg = {
   943	        temperature: Math.round((sum.temp / count) * 10) / 10,
   944	        pressure: Math.round((sum.press / count) * 100) / 100,
   945	        humidity: Math.round((sum.humid / count) * 10) / 10
   946	    };
   947	    latestSensor = avg;
   948	    const faults = analyzeFault(avg);
   949	    broadcastSensorData(avg, faults, { source: 'fallback' });
   950	    if (faults.length > 0) {
   951	        await handleAlarm(faults, avg);
   952	        console.log(`🚨 [旧固件回退] 检测到 ${faults.length} 个故障`);
   953	    }
   954	    sensorBuffer = [];
   955	}

   968	        let humidity = Math.round((20 + Math.random() * 75) * 10) / 10;

   972	            if (choice === 0) temp = 80 + Math.random() * 20;
   973	            else if (choice === 1) pressure = 1000 + Math.random() * 300;
   974	            else humidity = 85 + Math.random() * 15;
   975	        }
   976	
   977	        const avg = {
   978	            temperature: Math.round(temp * 10) / 10,

   981	        };
   982	        latestSensor = avg;

   984	        broadcastSensorData(avg, faults, { source: 'simulation' });
